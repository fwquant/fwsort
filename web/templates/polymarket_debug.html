# F3 市价单：1 USDC 买 UP result = await pm.下单( 标的代码="btc-updown-5m-{epoch}", outcome="UP", amount=Decimal(1), side="BUY", ) # F3 市价单：1 USDC 买 DOWN result = await pm.下单( 标的代码="btc-updown-4h-{epoch}", outcome="DOWN", amount=Decimal(1), side="BUY", ) # 卖出持仓 result = await pm.下单( 标的代码="btc-updown-5m-{epoch}", outcome="UP", shares=Decimal(0.5), side="SELL", )
